2. HMAC-Authentifizierung implementieren
Der Wawi-Proviantomat Agent muss diese Authentifizierung implementieren:

const crypto = require('crypto');
function verifyHMACSignature(req, res, next) {
  const signature = req.headers.authorization?.replace('Bearer ', '');
  const timestamp = req.headers['x-timestamp'];
  const appSource = req.headers['x-app-source'];
  const apiKey = req.headers['x-api-key'];
  
  // Payload rekonstruieren
  const payload = `${req.method}:${req.path}:${JSON.stringify(req.body)}:${timestamp}:${appSource}`;
  const expectedSignature = crypto.createHmac('sha256', process.env.INTER_APP_SECRET).update(payload).digest('hex');
  
  if (signature !== expectedSignature) {
    return res.status(401).json({ success: false, error: 'Invalid signature' });
  }
  
  next();
}
