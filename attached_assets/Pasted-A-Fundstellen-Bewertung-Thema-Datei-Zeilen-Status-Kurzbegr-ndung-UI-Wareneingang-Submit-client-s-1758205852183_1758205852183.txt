+export const fifoDepleteSchema = z.object({
+  warehouseId: z.number().int().positive(),
+  productId: z.number().int().positive(),
+  quantity: z.number().positive(),
+  referenceType: z.enum(["REFILL", "TRANSFER", "ADJUSTMENT"]),
+  referenceId: z.string().min(1),
+  targetLocation: z.string().min(1),
+  actorId: z.number().int().positive().nullable(),
+});
+
+export type FifoDepleteInput = z.infer<typeof fifoDepleteSchema>;
+
+export async function fifoDeplete(input: FifoDepleteInput) {
+  const parsed = fifoDepleteSchema.parse(input);
+  return db.transaction(async (tx) => {
+    let remaining = parsed.quantity;
+    const consumed: Array<{ batchId: number; quantity: number }> = [];
+
+    const inventoryRow = await tx.execute(
+      sql`SELECT id, quantity FROM inventory_items WHERE warehouse_id = ${parsed.warehouseId} AND product_id = ${parsed.productId} FOR UPDATE`
+    );
+    const beforeTotal = inventoryRow.rows[0]?.quantity ?? 0;
+    if (beforeTotal < parsed.quantity) {
+      throw new Error("Nicht genügend Bestand vorhanden");
+    }
+
+    const batches = await tx.execute(
+      sql`SELECT id, current_quantity, expiry_date FROM product_batches
+          WHERE warehouse_id = ${parsed.warehouseId}
+            AND product_id = ${parsed.productId}
+            AND current_quantity > 0
+          ORDER BY expiry_date ASC, received_date ASC, id ASC
+          FOR UPDATE`
+    );
+
+    for (const batch of batches.rows) {
+      if (remaining <= 0) break;
+      const take = Math.min(remaining, batch.current_quantity);
+      await tx.execute(
+        sql`UPDATE product_batches SET current_quantity = current_quantity - ${take}, updated_at = NOW()
+            WHERE id = ${batch.id}`
+      );
+      consumed.push({ batchId: batch.id, quantity: take });
+      remaining -= take;
+
+      await tx.insert(inventoryMovements).values({
+        warehouseId: parsed.warehouseId,
+        productId: parsed.productId,
+        batchId: batch.id,
+        quantity: -take,
+        movementType: "FILL",
+        direction: "OUT",
+        referenceType: parsed.referenceType,
+        referenceId: parsed.referenceId,
+        notes: parsed.targetLocation,
+        performedBy: parsed.actorId,
+        performedAt: sql`NOW()`,
+        previousStock: null,
+        currentStock: null,
+        createdAt: sql`NOW()`,
+        updatedAt: sql`NOW()`,
+      });
+    }
+
+    if (remaining > 0) {
+      throw new Error("FIFO-Entnahme unvollständig, Charge fehlt");
+    }
+
+    await tx.execute(
+      sql`UPDATE inventory_items SET quantity = quantity - ${parsed.quantity}, updated_at = NOW()
+          WHERE warehouse_id = ${parsed.warehouseId} AND product_id = ${parsed.productId}`
+    );
+
+    const afterTotal = beforeTotal - parsed.quantity;
+    return {
+      consumed,
+      beforeTotal,
+      afterTotal,
+    };
+  });

import { beforeEach, describe, expect, it } from "vitest";

let db: ReturnType<typeof drizzle>;

beforeEach(async () => {
  pool = new Pool({ connectionString: process.env.TEST_DATABASE_URL });

  await db.insert(schema.inventoryItems).values({
    warehouseId: 1,
    productId: 10,
    quantity: 30,
    status: "active",
  });
  await db.insert(schema.productBatches).values([
    { warehouseId: 1, productId: 10, batchNumber: "FIFO-1", initialQuantity: 10, currentQuantity: 10, expiryDate: new Date("2025-01-01"), receivedDate: new Date("2024-10-01"), status: "active" },
    { warehouseId: 1, productId: 10, batchNumber: "FIFO-2", initialQuantity: 20, currentQuantity: 20, expiryDate: new Date("2025-02-01"), receivedDate: new Date("2024-11-01"), status: "active" },
  ]);
});

describe("fifoDeplete", () => {
  it("zieht chargengenau nach FIFO ab", async () => {
    const result = await fifoDeplete({
      warehouseId: 1,
      productId: 10,
      quantity: 15,
      referenceType: "REFILL",
      referenceId: "refill-1",
      targetLocation: "machine:5",
      actorId: 7,
    });
    expect(result.consumed).toEqual([
      { batchId: expect.any(Number), quantity: 10 },
      { batchId: expect.any(Number), quantity: 5 },
    ]);
    expect(result.afterTotal).toBe(15);
  });

  it("verhindert negative Bestände", async () => {
    await expect(fifoDeplete({
      warehouseId: 1,
      productId: 10,
      quantity: 100,
      referenceType: "REFILL",
      referenceId: "refill-2",
      targetLocation: "machine:5",
      actorId: 7,
    })).rejects.toThrow(/Nicht genügend Bestand/);
  });

  it("ist race-safe (FOR UPDATE)", async () => {
    const [r1, r2] = await Promise.allSettled([
      fifoDeplete({ warehouseId: 1, productId: 10, quantity: 25, referenceType: "REFILL", referenceId: "r1", targetLocation: "machine:A", actorId: 1 }),
      fifoDeplete({ warehouseId: 1, productId: 10, quantity: 10, referenceType: "REFILL", referenceId: "r2", targetLocation: "machine:B", actorId: 2 }),
    ]);
    const successes = [r1, r2].filter(r => r.status === "fulfilled");
    expect(successes).toHaveLength(1);
  });
});
