2. Lagerbestand-Verwaltung
Dual-System für Bestände:

[inventory_items]: Normaler Bestand ohne MHD

FIFO-Logik:

SELECT * FROM product_batches 
WHERE productId = ? AND warehouseId = ? AND status = 'active'
ORDER BY expiryDate ASC, receivedDate ASC

E-Mail-Benachrichtigungen:
Tägliche MHD-Berichte:

// dailyEmailService.ts sendet automatische Warnungen
const mhdWarnings = await db.query(`
  SELECT pb.*, p.product_name, w.name as warehouse_name
  FROM product_batches pb
  JOIN products p ON pb.product_id = p.id  
  JOIN warehouses w ON pb.warehouse_id = w.id
  WHERE pb.expiry_date <= CURRENT_DATE + INTERVAL '7 days'
  AND pb.current_quantity > 0
`);
// E-Mail-Template mit MHD-Warnungen
const emailContent = `
  <h3>⚠️ MHD-Warnungen</h3>
  ${mhdWarnings.map(batch => `

    <p>MHD: ${batch.expiry_date} (${batch.days_until_expiry} Tage)</p>
    <p>Lager: ${batch.warehouse_name}, Menge: ${batch.current_quantity}</p>
  `).join('')}
`;
