function centsToEuro(v?: number | null): number | null {
  return (typeof v === 'number') ? Math.round(v) / 100 : null;
}

function toTZ(dateStr: string): string {
  // API liefert "local naive"; wir speichern als timestamptz (Europe/Berlin).
  // Wenn ihr serverseitig UTC wollt: hier explizit konvertieren.
  return `${dateStr}+01:00`; // minimal-invasiv; besser: luxon/dayjs+tz einsetzen
}

export function mapVendonTransaction(tx: VendonTx) {
  return {
    vendon_id: tx.id,
    datetime: toTZ(tx.datetime),
    machine_id: tx.machine_id ?? null,
    machine_name: tx.machine_name ?? null,
    location_id: tx.location_id ?? null,
    location_name: tx.location_name ?? null,
    product_id: tx.product_id ?? null,
    product_name: tx.product_name ?? null,
    price: centsToEuro(tx.product_price),
    cost: centsToEuro(tx.product_cost), // bleibt null, wenn API nichts liefert
    quantity: tx.quantity ?? 1,
    total_amount: centsToEuro(tx.total_amount),
    payment_method: tx.payment_method ?? null,
    transaction_type: tx.transaction_type ?? null,
    currency: tx.currency ?? 'EUR',
    status: tx.status ?? null,
    refund_reason: tx.refund_reason ?? null,

// server/lib/retry.ts
export async function retryWithBackoff<T>(
  op: () => Promise<T>,
  tries = 3,
  baseMs = 1000
): Promise<T> {
  let lastErr: any;
  for (let i = 0; i < tries; i++) {
    try { return await op(); } 
    catch (err: any) {
      lastErr = err;
      const status = err?.response?.status;
      // Keine Retries bei 4xx außer 408/429
      if (status && status >= 400 && status < 500 && status !== 408 && status !== 429) break;
      const delay = baseMs * Math.pow(2, i);
      await new Promise(r => setTimeout(r, delay));
    }
  }
  throw lastErr;
}

Automatisierte Coverage‑Prüfung (API‑Keys → DB‑Spalten)

Dieses Skript zieht ein Sample aus der Vendon‑API, extrahiert die Keys und vergleicht sie mit information_schema.columns eurer DB. Ausgegeben werden: abgedeckte Felder, fehlende Spalten, gemappte aber leere Felder.

// scripts/check_vendon_api_db_coverage.ts
import pg from 'pg';
import { getTransactions } from '../server/services/vendonAPI';

async function main() {
  const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });
  const client = await pool.connect();

  try {
    const sample = await getTransactions({ limit: 50 }); // kleine Stichprobe
    const apiKeys = new Set<string>();
    for (const tx of sample) Object.keys(tx).forEach(k => apiKeys.add(k));

    const { rows } = await client.query(`
      SELECT column_name 
      FROM information_schema.columns 
      WHERE table_name='transactions'
    `);
    const dbCols = new Set(rows.map(r => r.column_name));

    // simple Heuristik: snake_case der API‑Keys
    const normalize = (k: string) => k.replace(/[A-Z]/g, m => `_${m.toLowerCase()}`);

    const missingInDB: string[] = [];
    apiKeys.forEach(k => {
      const col = normalize(k);
      if (!dbCols.has(col)) missingInDB.push(`${k} → ${col}`);
    });

    console.log('API Keys:', [...apiKeys].sort());
    console.log('DB Columns:', [...dbCols].sort());
    console.log('Missing columns (proposed mapping):', missingInDB);

  } finally {
    client.release();
  }
}

main().catch(e => { console.error(e); process.exit(1); });

Pseudocode (cron alle 5 min):

const r = await db.one(`
  SELECT
    SUM(errors) AS errors,
    SUM(items_found) AS found,
    SUM(items_saved) AS saved
  FROM sync_logs
  WHERE created_at >= NOW() - INTERVAL '15 min'
`);
const ok = (Number(r.saved) / Math.max(1, Number(r.found))) >= 0.9 && Number(r.errors) === 0;
