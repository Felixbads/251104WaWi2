// Hilfsfunktion zum Auslesen des ersten belegten Query-Parameters
const getQueryParam = (req: Request, names: string[]): string | undefined => {
  for (const name of names) {
    const value = (req.query as Record<string, unknown>)[name];
    if (value === undefined || value === null) {
      continue;
    }

    if (Array.isArray(value)) {
      if (value.length === 0) {
        continue;
      }
      const first = value[0];
      if (first === undefined || first === null) {
        continue;
      }
      return String(first);
    }

    return String(value);
  }

  return undefined;
};
