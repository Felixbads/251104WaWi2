
# LAGER-KONZEPT VOLLSTÄNDIGE IMPLEMENTIERUNG

## ÜBERSICHT DER AKTUELLEN SITUATION

**Status**: Das Lagersystem ist grundlegend implementiert, aber **kritische Lücken** bestehen bei der MHD-Verfolgung und FIFO-Integration vom Lager zu den Automaten.

### WAS FUNKTIONIERT BEREITS:
1. ✅ **Warehouse3-Grundstruktur**: `server/warehouse3.storage.ts` - Vollständiges CRUD für Lager, Zuordnungen, Inventar
2. ✅ **MHD-FIFO Service**: `server/services/mhdFifoService.ts` - FIFO-Logik für Chargen implementiert
3. ✅ **Batch-Management**: `inventory_batches` Tabelle mit MHD-Tracking
4. ✅ **UI-Komponenten**: Warehouse3Dashboard, Batch-Dialoge, Inventory-Management

### KRITISCHE LÜCKEN:
1. ❌ **Refill-MHD-Transfer**: MHD wird NICHT vom Lager zum Automaten übertragen
2. ❌ **Machine-Stock-MHD**: `machine_stocks` Tabelle fehlt MHD-Felder
3. ❌ **FIFO-Integration**: Refill-Prozess nutzt FIFO nicht automatisch
4. ❌ **MHD-Verfolgung**: Kein durchgängiges MHD-Tracking vom Wareneingang bis Verkauf

```typescript
// NEUE FUNKTION: FIFO Batch Allocation für Machine Stock
async function calculateFifoBatchAllocation(
  db: any, 
  machineId: number, 
  productName: string, 
  currentQuantity: number
): Promise<{
  batches: Array<{
    batchId: number;
    batchNumber: string;
    quantity: number;
    expiryDate: string;
    daysUntilExpiry: number;
  }>;
  earliestMhd: string | null;
  totalBatches: number;
  mhdStatus: 'ok' | 'warning' | 'expired';
}> {
  // FIFO-Batch-Zuordnung basierend auf machine_stocks
  const batchResult = await db.query(`
    SELECT 
      ms.batch_id,
      ms.expiry_date,
      ms.quantity,
      ib.batch_number,
      DATE_PART('day', ms.expiry_date - CURRENT_DATE) as days_until_expiry
    FROM machine_stocks ms
    LEFT JOIN inventory_batches ib ON ms.batch_id = ib.id
    WHERE ms.machine_id = $1 
      AND ms.product_vendon_id IN (
        SELECT vendon_id FROM products WHERE product_name ILIKE $2
      )
      AND ms.quantity > 0
    ORDER BY ms.expiry_date ASC NULLS LAST, ms.received_date ASC
  `, [machineId, `%${productName}%`]);

  // Verarbeitung und Status-Berechnung
  const batches = batchResult.rows.map(row => ({
    batchId: row.batch_id,
    batchNumber: row.batch_number || `BATCH-${row.batch_id}`,
    quantity: row.quantity,
    expiryDate: row.expiry_date,
    daysUntilExpiry: row.days_until_expiry || 999
  }));

  const earliestMhd = batches.length > 0 ? batches[0].expiryDate : null;
  const mhdStatus = batches.some(b => b.daysUntilExpiry < 0) ? 'expired' :
                   batches.some(b => b.daysUntilExpiry <= 7) ? 'warning' : 'ok';

  return {
    batches,
    earliestMhd,
    totalBatches: batches.length,
    mhdStatus
  };
}
```

```typescript
const MHDBatchDisplay = ({ stock }: { stock: MachineStock }) => {
  if (!stock.batches || stock.batches.length === 0) {
    return <Badge variant="outline">Kein MHD</Badge>;
  }

  const earliestBatch = stock.batches[0];
  const statusColor = stock.mhdStatus === 'expired' ? 'destructive' : 
                     stock.mhdStatus === 'warning' ? 'secondary' : 'default';

  return (
    <div className="space-y-1">
      <Badge variant={statusColor}>
        MHD: {format(new Date(earliestBatch.expiryDate), 'dd.MM.yyyy')}
      </Badge>
      {stock.batches.length > 1 && (
        <Badge variant="outline">+{stock.batches.length - 1} weitere</Badge>
      )}
    </div>
  );
};
```

```typescript
// NEUE FUNKTION: MHD-basierte Template-Optimierung
export async function optimizeTemplateForShortExpiry(
  templateId: number,
  maxDaysUntilExpiry: number = 14
): Promise<{
  success: boolean;
  adjustedProducts: number;
  template?: MHDOptimizedTemplate;
}> {
  // 1. Template laden
  const template = await getRefillTemplate(templateId);
  if (!template) return { success: false, adjustedProducts: 0 };

  // 2. Für jedes Produkt MHD-Status prüfen
  let adjustedProducts = 0;
  const optimizedProducts = [];

  for (const product of template.products) {
    // Prüfe kürzeste MHD im Lager
    const batchResult = await db.query(`
      SELECT MIN(expiry_date) as earliest_expiry, SUM(quantity) as total_stock
      FROM inventory_batches ib
      JOIN products p ON ib.product_id = p.id  
      WHERE p.vendon_id = $1 AND ib.quantity > 0 AND ib.status = 'active'
    `, [product.vendonId]);

    const earliestExpiry = batchResult.rows[0]?.earliest_expiry;
    const daysUntilExpiry = earliestExpiry ? 
      Math.ceil((new Date(earliestExpiry).getTime() - Date.now()) / (1000 * 60 * 60 * 24)) : 999;

    let adjustedQuantity = product.quantity;
    let adjustmentReason = '';

    // MHD-basierte Mengenanpassung
    if (daysUntilExpiry <= 3) {
      adjustedQuantity = Math.min(product.quantity, 5); // Nur minimale Menge
      adjustmentReason = 'Kritisch: MHD in 3 Tagen - Minimalmenge';
      adjustedProducts++;
    } else if (daysUntilExpiry <= 7) {
      adjustedQuantity = Math.floor(product.quantity * 0.6); // 60% der ursprünglichen Menge
      adjustmentReason = 'Warnung: MHD in 1 Woche - Reduzierte Menge';
      adjustedProducts++;
    } else if (daysUntilExpiry <= maxDaysUntilExpiry) {
      adjustedQuantity = Math.floor(product.quantity * 0.8); // 80% der ursprünglichen Menge
      adjustmentReason = 'Vorsicht: Kurzes MHD - Leicht reduziert';
      adjustedProducts++;
    }

    optimizedProducts.push({
      ...product,
      originalQuantity: product.quantity,
      adjustedQuantity,
      mhdDays: daysUntilExpiry,
      riskCategory: daysUntilExpiry <= 3 ? 'critical' as const :
                   daysUntilExpiry <= 7 ? 'high' as const :
                   daysUntilExpiry <= 14 ? 'medium' as const : 'low' as const,
      adjustmentReason
    });

    // Template aktualisieren
    if (adjustedQuantity !== product.quantity) {
      await db.update(refillTemplateProducts)
        .set({ quantity: adjustedQuantity })
        .where(eq(refillTemplateProducts.id, product.id));
    }
  }

  return {
    success: true,
    adjustedProducts,
    template: {
      template,
      products: optimizedProducts,
      summary: {
        totalProducts: optimizedProducts.length,
        adjustedProducts,
        criticalProducts: optimizedProducts.filter(p => p.riskCategory === 'critical').length,
        totalReduction: optimizedProducts.reduce((sum, p) => sum + (p.originalQuantity - p.adjustedQuantity), 0),
        averageConfidence: 85 // MHD-basierte Optimierung hat hohe Konfidenz
      }
    }
  };
}
```

```typescript
// MHD-optimierte Refill-Vorlage
router.post('/:id/refill-templates/mhd-optimize', async (req, res) => {
  const machineId = parseInt(req.params.id);
  const { templateId, maxDaysUntilExpiry = 14 } = req.body;

  const result = await optimizeTemplateForShortExpiry(templateId, maxDaysUntilExpiry);
  
  if (result.success) {
    res.json({
      success: true,
      message: `Template optimiert: ${result.adjustedProducts} Produkte angepasst`,
      template: result.template
    });
  } else {
    res.status(500).json({ error: 'Optimierung fehlgeschlagen' });
  }
});

// FIFO-Batch-Details für Automaten-Bestand
```
