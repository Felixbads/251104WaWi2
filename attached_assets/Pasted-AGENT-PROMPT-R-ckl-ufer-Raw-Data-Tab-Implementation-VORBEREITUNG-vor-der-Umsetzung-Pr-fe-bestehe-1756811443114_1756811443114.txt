  r.created_at as refill_date,
  m.name as machine_name,
  r.machine_id,
  rd.product_name,
  r.operator,
  rd.removed as removed_quantity,
  (rd.removed * p.purchase_price) as estimated_loss
FROM refills r
JOIN refill_details rd ON r.id = rd.refill_id  
JOIN machines m ON r.machine_id = m.machine_id
LEFT JOIN products p ON rd.product_name = p.name
WHERE rd.removed > 0
  AND r.created_at BETWEEN ? AND ?
  [OPTIONAL FILTERS]
ORDER BY r.created_at DESC
LIMIT ? OFFSET ?
